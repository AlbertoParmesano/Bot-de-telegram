// Inicializar Telegram Web App
const tg = window.Telegram.WebApp;
let messageCount = 0;
let actionCount = 0;

// Función inicialización
function initTelegramWebApp() {
    console.log('Inicializando Telegram Web App...');
    
    // Expandir la aplicación para usar toda la pantalla
    tg.expand();
    
    // Configurar color de la barra superior
    tg.setHeaderColor('#0088cc');
    tg.setBackgroundColor('#f5f5f5');
    
    // Obtener información del usuario
    const user = tg.initDataUnsafe?.user;
    if (user) {
        displayUserInfo(user);
    }
    
    // Mostrar botón principal si estamos en Telegram
    if (tg.platform !== 'unknown') {
        tg.MainButton.setText('Cerrar Interfaz')
            .setParams({ color: '#e74c3c' })
            .onClick(closeWebApp)
            .show();
    }
    
    // Actualizar estadísticas
    updateStats();
    
    // Mensaje de bienvenida en consola
    addConsoleMessage('✅ Interfaz web cargada correctamente');
    addConsoleMessage('👤 Usuario detectado: ' + (user ? user.username : 'Invitado'));
    addConsoleMessage('📱 Plataforma: ' + tg.platform);
    
    // Eventos de Telegram
    tg.onEvent('viewportChanged', (data) => {
        addConsoleMessage('📐 Viewport cambiado: ' + JSON.stringify(data));
    });
    
    tg.onEvent('themeChanged', () => {
        addConsoleMessage('🎨 Tema cambiado: ' + tg.themeParams);
    });
}

// Mostrar información del usuario
function displayUserInfo(user) {
    const userInfoElement = document.getElementById('user-info');
    if (userInfoElement) {
        userInfoElement.innerHTML = 
            <i class="fas fa-user"></i>
            <span>${user.first_name} ${user.last_name || ''} @${user.username || 'sin_usuario'}</span>
        ;
    }
}

// Enviar mensaje al bot
function sendMessage() {
    const input = document.getElementById('message-input');
    const message = input.value.trim();
    
    if (!message) {
        addConsoleMessage('❌ Error: El mensaje está vacío');
        showNotification('Por favor, escribe un mensaje', 'error');
        return;
    }
    
    // Simular envío (en producción, aquí llamarías a tu API)
    addConsoleMessage(📤 Enviando mensaje: "${message}");
    showNotification('Mensaje enviado correctamente', 'success');
    
    // Si estamos en Telegram Web App, podemos enviar datos al bot
    if (tg.platform !== 'unknown') {
        tg.sendData(JSON.stringify({
            action: 'message',
            content: message,
            timestamp: new Date().toISOString()
        }));
    }
    
    // Actualizar estadísticas
    messageCount++;
    updateStats();
    
    // Limpiar input
    input.value = '';
    input.focus();
}

// Enviar acción predefinida
function sendAction(actionType) {
    const actions = {
        saludo: { text: '¡Hola! 👋', message: 'El usuario envió un saludo' },
        hora: { text: 🕒 Hora actual: ${new Date().toLocaleTimeString()}, message: 'El usuario solicitó la hora' },
        datos: { text: '📊 Mostrando datos del usuario...', message: 'El usuario solicitó sus datos' }
    };
    
    const action = actions[actionType];
    if (!action) return;
    
    addConsoleMessage(⚡ Acción ejecutada: ${action.message});
    showNotification(action.text, 'info');
    
    // Enviar a Telegram si es posible
    if (tg.platform !== 'unknown') {
        tg.sendData(JSON.stringify({
            action: actionType,
            timestamp: new Date().toISOString(),
            source: 'web_interface'
        }));
    }
    
    // Actualizar estadísticas
    actionCount++;
    updateStats();
}

// Ejecutar comando desde consola
function executeCommand() {
    const input = document.getElementById('console-input');
    const command = input.value.trim().toLowerCase();
    
    if (!command) return;addConsoleMessage($ ${command});
    
    // Comandos disponibles
    const commands = {
        help: 'Comandos disponibles: help, clear, stats, time, user',
        clear: () => {
            document.getElementById('console-output').innerHTML = '';
            addConsoleMessage('Consola limpiada');
        },
        stats: () => {
            addConsoleMessage(📊 Estadísticas: ${messageCount} mensajes, ${actionCount} acciones);
        },
        time: () => {
            addConsoleMessage(🕒 Hora del servidor: ${new Date().toISOString()});
        },
        user: () => {
            const user = tg.initDataUnsafe?.user;
            if (user) {
                addConsoleMessage(👤 Usuario: ${user.first_name} (ID: ${user.id}));
            } else {
                addConsoleMessage('👤 Usuario no disponible');
            }
        }
    };
    
    if (commands[command]) {
        if (typeof commands[command] === 'function') {
            commands[command]();
        } else {
            addConsoleMessage(commands[command]);
        }
    } else {
        addConsoleMessage(❌ Comando no reconocido: "${command}");
    }
    
    input.value = '';
    input.focus();
}

// Actualizar estadísticas en la UI
function updateStats() {
    document.getElementById('stat-messages').textContent = messageCount;
    document.getElementById('stat-actions').textContent = actionCount;
}

// Añadir mensaje a la consola
function addConsoleMessage(message) {
    const consoleOutput = document.getElementById('console-output');
    const messageElement = document.createElement('div');
    messageElement.className = 'console-line';
    messageElement.textContent = > ${message};
    consoleOutput.appendChild(messageElement);
    consoleOutput.scrollTop = consoleOutput.scrollHeight;
}

// Mostrar notificación
function showNotification(message, type = 'info') {
    // Aquí podrías implementar un sistema de notificaciones
    console.log([${type.toUpperCase()}] ${message});
    
    // Ejemplo simple usando alerta de Telegram
    if (tg.platform !== 'unknown') {
        tg.showPopup({
            title: type === 'error' ? 'Error' : 
                   type === 'success' ? 'Éxito' : 'Información',
            message: message,
            buttons: [{ type: 'ok' }]
        });
    }
}

// Cerrar Web App
function closeWebApp() {
    if (tg.platform !== 'unknown') {
        tg.close();
    } else {
        alert('Esta funcionalidad solo está disponible en la app de Telegram');
    }
}

// Refrescar datos
function refreshData() {
    addConsoleMessage('🔄 Actualizando datos...');
    // Simular carga
    setTimeout(() => {
        addConsoleMessage('✅ Datos actualizados');
        showNotification('Datos actualizados correctamente', 'success');
    }, 1000);
}

// Mostrar ayuda
function showHelp() {
    const helpText = 
    🤖 **Ayuda de la Interfaz Web**
    
    **Funcionalidades:**
    1. Enviar mensajes al bot
    2. Acciones rápidas predefinidas
    3. Ver estadísticas de uso
    4. Consola interactiva
    
    **Comandos de consola:**
    - help: Muestra esta ayuda
    - clear: Limpia la consola
    - stats: Muestra estadísticas
    - time: Muestra la hora actual
    - user: Muestra info del usuario
    
    **Nota:** Esta interfaz está optimizada para usar dentro de la app de Telegram.
    ;
    
    showNotification(helpText, 'info');
}

// Manejar tecla Enter en inputs
document.addEventListener('DOMContentLoaded', function() {
    initTelegramWebApp();
    
    // Enter en input de mensaje
    document.getElementById('message-input').addEventListener('keypress', function(e) {
        if (e.key === 'Enter') sendMessage();
    });
    
    // Enter en consola
    document.getElementById('console-input').addEventListener('keypress', function(e) {
        if (e.key === 'Enter') executeCommand();
    });
});
