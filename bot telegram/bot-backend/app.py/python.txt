import os
import logging
from flask import Flask, request, jsonify, render_template
from telegram import Bot, Update, InlineKeyboardButton, InlineKeyboardMarkup, WebAppInfo
from telegram.ext import Application, CommandHandler, ContextTypes, MessageHandler, filters
import json
from dotenv import load_dotenv

# Cargar variables de entorno
load_dotenv()

# Configurar logging
logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    level=logging.INFO
)

app = Flask(name)

# Configuración
TOKEN = os.getenv('TELEGRAM_BOT_TOKEN', 'TU_TOKEN_AQUI')
WEB_APP_URL = os.getenv('WEB_APP_URL', 'https://tu-app-en-render.onrender.com/web-interface')
PORT = int(os.getenv('PORT', 5000))

# Crear aplicación de Telegram
telegram_app = Application.builder().token(TOKEN).build()

# ================= RUTAS WEB (FLASK) =================
@app.route('/')
def home():
    return "¡Servidor del Bot funcionando! 🚀"

@app.route('/web-interface')
def web_interface():
    return render_template('index.html')

@app.route('/api/send-message', methods=['POST'])
def api_send_message():
    """API para enviar mensajes desde la web al bot"""
    try:
        data = request.json
        chat_id = data.get('chat_id')
        message = data.get('message')
        
        if not chat_id or not message:
            return jsonify({'error': 'Faltan datos'}), 400
        
        # Aquí enviarías el mensaje usando el bot
        # Por simplicidad, solo lo guardamos en log
        logging.info(f"Mensaje para {chat_id}: {message}")
        
        return jsonify({
            'success': True,
            'message': 'Mensaje enviado (simulado)'
        })
    
    except Exception as e:
        logging.error(f"Error en API: {e}")
        return jsonify({'error': str(e)}), 500

# ================= MANEJADORES DE TELEGRAM =================
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Maneja el comando /start"""
    user = update.effective_user
    
    # Crear teclado con botón para Web App
    keyboard = [
        [InlineKeyboardButton(
            "🚀 Abrir Interfaz Web",
            web_app=WebAppInfo(url=WEB_APP_URL)
        )]
    ]
    reply_markup = InlineKeyboardMarkup(keyboard)
    
    await update.message.reply_html(
        f"Hola {user.mention_html()}! 👋\n\n"
        "Bienvenido a mi bot con interfaz web.\n"
        "Haz clic en el botón para abrir la interfaz:",
        reply_markup=reply_markup
    )

async def help_command(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Maneja el comando /help"""
    help_text = """
    🤖 *Comandos disponibles:*
    
    /start - Iniciar el bot y mostrar interfaz web
    /help - Mostrar esta ayuda
    /info - Información sobre el bot
    
    También puedes usar los botones en la interfaz web.
    """
    await update.message.reply_text(help_text, parse_mode='Markdown')

async def handle_web_app_data(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Recibe datos de la Web App"""
    try:
        data = json.loads(update.message.web_app_data.data)
        user = update.effective_user
        
        await update.message.reply_text(
            f"✅ Datos recibidos de la web:\n"
            f"📊 Acción: {data.get('action', 'Desconocida')}\n"
            f"🕒 Hora: {data.get('time', 'No especificada')}\n\n"
            f"Gracias por usar la interfaz web, {user.first_name}!"
        )
    except Exception as e:
        logging.error(f"Error procesando web app data: {e}")
        await update.message.reply_text("❌ Error procesando los datos.")

# ================= CONFIGURACIÓN =================
def setup_telegram_handlers():
    """Configura los manejadores de comandos de Telegram"""
    telegram_app.add_handler(CommandHandler("start", start))
    telegram_app.add_handler(CommandHandler("help", help_command))
    telegram_app.add_handler(MessageHandler(filters.StatusUpdate.WEB_APP_DATA, handle_web_app_data))
# ================= INICIALIZACIÓN =================
if name == 'main':
    # Configurar manejadores
    setup_telegram_handlers()
    
    # Iniciar el bot de Telegram en segundo plano
    import threading
    def run_telegram_bot():
        telegram_app.run_polling(drop_pending_updates=True)
    
    telegram_thread = threading.Thread(target=run_telegram_bot, daemon=True)
    telegram_thread.start()
    
    # Iniciar servidor Flask
    app.run(host='0.0.0.0', port=PORT, debug=False)
